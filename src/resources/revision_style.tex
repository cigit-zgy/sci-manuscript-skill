% Added text color.
\definecolor{RevisionAddedColor}{RGB}{0,92,153}

% Added text underline style. Alternatives include \uline and identity (#1).
% The runtime loads xeCJKfntef for CJK projects, which patches ulem so wavy
% and strike lines can break across CJK text while staying compatible with
% embedded natbib citations; plain ulem suffices for Latin scripts.
\newcommand{\RevisionAddedUnderline}[1]{\uwave{#1}}

% =====================================
% Deleted text
% =====================================

% Deleted text color.
\definecolor{RevisionDeletedColor}{RGB}{220,45,45}

% Deleted text strikeout style. Use identity (#1) to disable the strikeout.
\newcommand{\RevisionDeletedStrikeout}[1]{\sout{#1}}

\newcommand{\RevisionSelfBackground}[1]{#1}
