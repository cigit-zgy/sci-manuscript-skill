% Current-only highlighted-revision primitives. The renderer never emits
% parent-only deletion content.

\RequirePackage{xcolor}
\RequirePackage{lineno}
\AtBeginDocument{\linenumbers}

% Thin package-owned compiled-state proof. It records schema execution only;
% scientific prose and revision content remain source-owned.
\newwrite\SCIStateStream
\AtBeginDocument{%
  \immediate\openout\SCIStateStream=\jobname.sci
  \immediate\write\SCIStateStream{SCI_SCHEMA|1}%
  \immediate\write\SCIStateStream{DOCUMENT|marked}%
  \immediate\write\SCIStateStream{MARKED_SCHEMA|1}}
\AtEndDocument{\immediate\closeout\SCIStateStream}

\makeatletter
\newcommand{\SCISetCitationColor}[1]{%
  \@ifpackageloaded{hyperref}{\hypersetup{citecolor=#1}}{}%
}
\newcommand{\SCIEmitRevision}[2]{%
  \@ifundefined{SCIRevisionSeen@#1}{%
    \expandafter\gdef\csname SCIRevisionSeen@#1\endcsname{1}%
    \immediate\write\SCIStateStream{REVISION|#1|#2}}{}}
\makeatother
