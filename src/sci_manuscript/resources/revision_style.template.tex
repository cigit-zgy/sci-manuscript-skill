% =====================================
% Deleted text
% =====================================

% Deleted text color.
\definecolor{RevisionDeletedColor}{RGB}{160,160,160}
\newcommand{\RevisionDeletionThickness}{0.8pt}

\newcommand{\RevisionDeletedStrikeout}[1]{%
  \ifcsname CJKsout\endcsname
    \CJKsout*[thickness=\RevisionDeletionThickness,format=\color{RevisionDeletedColor},textformat=\color{RevisionDeletedColor}]{#1}%
  \else\begingroup\renewcommand{\ULthickness}{\RevisionDeletionThickness}\sout{#1}\endgroup\fi}
